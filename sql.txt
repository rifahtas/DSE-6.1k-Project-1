   username    VARCHAR(50) PRIMARY KEY,
   password    VARCHAR(255) NOT NULL,
   firstname   VARCHAR(50),
   lastname    VARCHAR(50),
   salary      FLOAT,
   age         INTEGER,
   registerday DATE,
   signintime  DATETIME NULL
);
